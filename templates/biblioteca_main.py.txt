def obter_pasta_dados_app():
    nome_pasta_app = "".join(
        caractere
        if caractere.isalnum()
        else "_"
        for caractere in NOME_APP
    ).strip("_").lower()

    pasta = (
        Path(
            App.get_running_app().user_data_dir
        )
        / nome_pasta_app
    )

    pasta.mkdir(
        parents=True,
        exist_ok=True
    )

    return pasta

    def selecionar_modulo(
        self,
        modulo
    ):
        if modulo == "Categorias":
            self.abrir_categorias()
            return

        if modulo == "Documentos":
            self.abrir_documentos()
            return

        if modulo == "PDFs":
            self.abrir_pdfs()
            return

        if modulo == "Imagens":
            self.abrir_imagens()
            return

        if modulo == "Pesquisa":
            self.abrir_pesquisa()
            return

        if modulo == "Favoritos":
            self.abrir_favoritos()
            return

        self.status.text = (
            "Módulo selecionado: "
            + modulo
        )


    def carregar_favoritos(self):
        if not self.arquivo_favoritos.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_favoritos.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_favoritos(self, favoritos):
        self.arquivo_favoritos.write_text(
            json.dumps(
                favoritos,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def esta_favorito(
        self,
        origem,
        item_id
    ):
        return any(
            item.get(
                "origem"
            ) == origem
            and item.get(
                "item_id"
            ) == item_id
            for item in self.carregar_favoritos()
        )

    def alternar_favorito(
        self,
        origem,
        item
    ):
        favoritos = (
            self.carregar_favoritos()
        )

        item_id = item.get(
            "id"
        )

        existente = None

        for favorito in favoritos:
            if (
                favorito.get(
                    "origem"
                ) == origem
                and favorito.get(
                    "item_id"
                ) == item_id
            ):
                existente = favorito
                break

        if existente is not None:
            favoritos.remove(
                existente
            )

            self.salvar_favoritos(
                favoritos
            )

            return False

        favoritos.append({
            "origem":
                origem,

            "item_id":
                item_id,

            "tipo":
                item.get(
                    "tipo",
                    (
                        "Imagem"
                        if origem == "imagens"
                        else "Documento"
                    )
                ),

            "titulo":
                item.get(
                    "titulo",
                    ""
                ),

            "categoria_nome":
                item.get(
                    "categoria_nome",
                    ""
                ),
        })

        self.salvar_favoritos(
            favoritos
        )

        return True

    def localizar_item_favorito(
        self,
        favorito
    ):
        origem = favorito.get(
            "origem"
        )

        item_id = favorito.get(
            "item_id"
        )

        if origem == "documentos":
            fonte = (
                self.carregar_documentos()
            )

        elif origem == "imagens":
            fonte = (
                self.carregar_imagens()
            )

        else:
            return None

        for item in fonte:
            if item.get(
                "id"
            ) == item_id:
                return item

        return None

    def abrir_item_favorito(
        self,
        favorito,
        ao_atualizar
    ):
        item = (
            self.localizar_item_favorito(
                favorito
            )
        )

        if item is None:
            self.status.text = (
                "O item original não foi encontrado."
            )
            return

        origem = favorito.get(
            "origem"
        )

        tipo = item.get(
            "tipo",
            ""
        )

        if (
            origem == "documentos"
            and tipo == "PDF"
        ):
            self.visualizar_pdf(
                item,
                ao_atualizar
            )
            return

        if origem == "documentos":
            self.visualizar_documento(
                item,
                ao_atualizar
            )
            return

        if origem == "imagens":
            self.visualizar_imagem(
                item,
                ao_atualizar
            )
            return

    def itens_pesquisaveis(self):
        itens = []

        for documento in self.carregar_documentos():
            copia = dict(
                documento
            )

            copia[
                "_origem"
            ] = "documentos"

            itens.append(
                copia
            )

        for imagem in self.carregar_imagens():
            copia = dict(
                imagem
            )

            copia[
                "_origem"
            ] = "imagens"

            itens.append(
                copia
            )

        return itens

    def normalizar_texto_pesquisa(
        self,
        valor
    ):
        texto = unicodedata.normalize(
            "NFKD",
            str(
                valor or ""
            )
        )

        texto = "".join(
            caractere
            for caractere in texto
            if not unicodedata.combining(
                caractere
            )
        )

        return " ".join(
            texto.casefold().split()
        )

    def pesquisar_itens(
        self,
        termo
    ):
        termo = (
            self.normalizar_texto_pesquisa(
                termo
            )
        )

        if not termo:
            return []

        resultados = []

        for item in self.itens_pesquisaveis():
            campos = [
                item.get(
                    "titulo",
                    ""
                ),

                item.get(
                    "categoria_nome",
                    ""
                ),

                item.get(
                    "autor",
                    ""
                ),

                item.get(
                    "referencia",
                    ""
                ),

                item.get(
                    "descricao",
                    ""
                ),

                item.get(
                    "tipo",
                    ""
                ),

                item.get(
                    "status",
                    ""
                ),
            ]

            texto_item = (
                self.normalizar_texto_pesquisa(
                    " ".join(
                        str(
                            campo or ""
                        )
                        for campo in campos
                    )
                )
            )

            if termo in texto_item:
                resultados.append(
                    item
                )

        return resultados

            self.carregar_imagens()

        ids = []

        for item in imagens:
            try:
                ids.append(
                    int(
                        item.get(

            ids,

            spacing=dp(6),

        )

        anterior = Button(
            text="ANTERIOR"
        )

        novo = Button(
            text="NOVO"
        )

        proximo = Button(
            text="PRÓXIMO"
        )

        navegacao.add_widget(
            anterior
        )

        navegacao.add_widget(
            novo
        )

        navegacao.add_widget(
            proximo
        )

        caixa.add_widget(
            navegacao
        )

        fechar = Button(
            text="FECHAR",
            size_hint_y=None,
            height=dp(48)
        )

        caixa.add_widget(
            fechar
        )

        popup = Popup(
            title="PDFs",
            content=caixa,
            size_hint=(0.96, 0.92),
            auto_dismiss=False
        )

        def atualizar():
            lista.clear_widgets()

            pdfs = (
                self.pdfs_registrados()
            )

            por_pagina = 4

            total_paginas = max(
                1,
                (
                    len(pdfs)
                    + por_pagina - 1
                ) // por_pagina
            )

            if (
                self.pagina_pdfs
                >= total_paginas
            ):
                self.pagina_pdfs = (
                    total_paginas - 1
                )

            inicio = (
                self.pagina_pdfs
                * por_pagina
            )

            fim = inicio + por_pagina

            pagina = pdfs[
                inicio:fim
            ]

            if not pagina:
                lista.add_widget(
                    Label(
                        text=(
                            "Nenhum PDF cadastrado."
                        )
                    )
                )

            for item in pagina:
                botao = Button(
                    text=(
                        item.get(
                            "titulo",
                            ""
                        )
                        + " - "
                        + item.get(
                            "status",
                            "Ativo"
                        )
                        + "\n"
                        + item.get(
                            "categoria_nome",
                            "Sem categoria"
                        )
                    ),
                    size_hint_y=None,
                    height=dp(70)
                )

                botao.bind(
                    on_release=lambda _b, d=item:
                    self.visualizar_pdf(
                        d,
                        atualizar
                    )
                )

                lista.add_widget(
                    botao
                )

            titulo.text = (
                "PDFs  "
                + str(
                    self.pagina_pdfs + 1
                )
                + "/"
                + str(total_paginas)
                + "   Total: "
                + str(
                    len(pdfs)
                )
            )

        def voltar(*_):
            if self.pagina_pdfs > 0:
                self.pagina_pdfs -= 1
                atualizar()

        def avancar(*_):
            pdfs = (
                self.pdfs_registrados()
            )

            total_paginas = max(
                1,
                (
                    len(pdfs) + 3
                ) // 4
            )

            if (
                self.pagina_pdfs + 1
                < total_paginas
            ):
                self.pagina_pdfs += 1
                atualizar()

        anterior.bind(
            on_release=voltar
        )

        proximo.bind(
            on_release=avancar
        )

        novo.bind(
            on_release=lambda *_:
            self.novo_pdf(
                atualizar
            )
        )

        fechar.bind(
            on_release=popup.dismiss
        )

        atualizar()
        popup.open()

    def carregar_documentos(self):
        if not self.arquivo_documentos.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_documentos.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_documentos(self, documentos):
        self.arquivo_documentos.write_text(
            json.dumps(
                documentos,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def proximo_id_documento(self):
        documentos = (
            self.carregar_documentos()
        )

        ids = []

        for item in documentos:
            try:
                ids.append(
                    int(
                        item.get(
                            "id",
                            0
                        )
                    )
                )
            except Exception:
                pass

        return max(
            ids,
            default=0
        ) + 1

    def carregar_categorias(self):
        if not self.arquivo_categorias.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_categorias.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_categorias(self, categorias):
        self.arquivo_categorias.write_text(
            json.dumps(
                categorias,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def proximo_id_categoria(self):
        categorias = (
            self.carregar_categorias()
        )

        ids = []

        for item in categorias:
            try:
                ids.append(
                    int(
                        item.get(
                            "id",
                            0
                        )
                    )
                )
            except Exception:
                pass

        return max(
            ids,
            default=0
        ) + 1

    def categorias_ativas(self):
        return [
            item
            for item in self.carregar_categorias()
            if item.get(
                "status",
                "Ativa"
            ) == "Ativa"
        ]

            ):
                pagina["numero"] = (
                    total_paginas - 1
                )

            inicio = (
                pagina["numero"]
                * por_pagina
            )

            fim = inicio + por_pagina

            atual = categorias[
                inicio:fim
            ]

            for item in atual:
                botao = Button(
                    text=item.get(
                        "nome",
                        ""
                    ),
                    size_hint_y=None,
                    height=dp(62)
                )

                botao.bind(
                    on_release=lambda _b, c=item:
                    selecionar(c)
                )

                lista.add_widget(
                    botao
                )

            titulo.text = (
                "CATEGORIAS  "
                + str(
                    pagina["numero"] + 1
                )
                + "/"
                + str(
                    total_paginas
                )
            )

        def voltar(*_):
            if pagina["numero"] > 0:
                pagina["numero"] -= 1
                atualizar()

        def avancar(*_):
            total_paginas = max(
                1,
                (
                    len(categorias) + 3
                ) // 4
            )

            if (
                pagina["numero"] + 1
                < total_paginas
            ):
                pagina["numero"] += 1
                atualizar()

        anterior.bind(
            on_release=voltar
        )

        proximo.bind(
            on_release=avancar
        )

        cancelar.bind(
            on_release=popup.dismiss
        )

        atualizar()
        popup.open()


class AplicativoGerado(App):

    def build(self):
        self.title = NOME_APP

        return BibliotecaApp()


if __name__ == "__main__":
    AplicativoGerado().run()
