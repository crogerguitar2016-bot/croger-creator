    def selecionar_modulo(self, modulo):
        if modulo == "Discípulos":
            self.abrir_discipulos()
            return

        if modulo == "Estudos":
            self.abrir_estudos()
            return

        if modulo == "Questões":
            self.abrir_questoes()
            return

        if modulo == "Respostas":
            self.abrir_respostas()
            return

        if modulo == "Presença":
            self.abrir_presencas()
            return

        if modulo == "Progresso":
            self.abrir_progresso()
            return

        if modulo == "Observações":
            self.abrir_observacoes()
            return

        self.status.text = (
            "Módulo selecionado: "
            + modulo
        )


    def carregar_observacoes(self):
        if not self.arquivo_observacoes.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_observacoes.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_observacoes(self, observacoes):
        self.arquivo_observacoes.write_text(
            json.dumps(
                observacoes,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def proximo_id_observacao(self):
        observacoes = self.carregar_observacoes()

        ids = []

        for item in observacoes:
            try:
                ids.append(
                    int(
                        item.get(
                            "id",
                            0
                        )
                    )
                )
            except Exception:
                pass

        return max(
            ids,
            default=0
        ) + 1

    def abrir_observacoes(self):

            spacing=dp(7),

            text="OBSERVAÇÕES",
            size_hint_y=None,

            observacoes = (
                self.carregar_observacoes()
            )

            total_paginas = max(
                1,
                (
                    len(observacoes)
                    + por_pagina - 1
                ) // por_pagina
            )

            if (
                self.pagina_observacoes
                >= total_paginas
            ):
                self.pagina_observacoes = (
                    total_paginas - 1
                )

                self.pagina_observacoes
                * por_pagina
            )

                inicio:fim
            ]

                lista.add_widget(
                    Label(
                        text=(

                        )
                    )
                )

                resumo = item.get(

                        resumo[:67]

                + "   Total: "
                + str(len(observacoes))

            if self.pagina_observacoes > 0:
                self.pagina_observacoes -= 1
                atualizar()

                self.carregar_observacoes()
            )

                1,
                (
                    len(observacoes) + 3
                ) // 4
            )

                self.pagina_observacoes + 1
                < total_paginas

                self.pagina_observacoes += 1
                atualizar()

            on_release=voltar
        )

        proximo.bind(
            on_release=avancar
        )

            on_release=lambda *_:
            self.nova_observacao(
                atualizar
            )
        )

        fechar.bind(
            on_release=popup.dismiss
        )

        atualizar()
        popup.open()

    def nova_observacao(

    ):

            text="ESCOLHER DISCÍPULO",

        )

        data = TextInput(
            hint_text="Data da observação",
            multiline=False,
            size_hint_y=None,
            height=dp(48)
        )

        observacao = TextInput(
            hint_text="Digite a observação",
            multiline=True,
            size_hint_y=None,
            height=dp(170)
        )

        caixa.add_widget(discipulo_texto)
        caixa.add_widget(escolher)
        caixa.add_widget(data)
        caixa.add_widget(observacao)

        aviso = Label(
            text="",
            size_hint_y=None,
            height=dp(32)
        )

        caixa.add_widget(aviso)

        botoes = BoxLayout(
            spacing=dp(8),
            size_hint_y=None,
            height=dp(50)
        )

        cancelar = Button(
            text="CANCELAR"
        )

        salvar = Button(
            text="SALVAR"
        )

        botoes.add_widget(cancelar)
        botoes.add_widget(salvar)

        caixa.add_widget(botoes)

        popup = Popup(
            title="Nova Observação",
            content=caixa,
            size_hint=(0.94, 0.90),
            auto_dismiss=False
        )

        def recebeu_discipulo(item):
            selecao["discipulo"] = item

            discipulo_texto.text = (
                item.get(
                    "nome",
                    ""
                )
            )

        escolher.bind(
            on_release=lambda *_:
            self.escolher_discipulo_observacao(
                recebeu_discipulo
            )
        )

        cancelar.bind(
            on_release=popup.dismiss
        )

        def confirmar(*_):
            discipulo = selecao[
                "discipulo"
            ]

            if discipulo is None:
                aviso.text = (
                    "Escolha o discípulo."
                )
                return

            data_texto = data.text.strip()

            if not data_texto:
                aviso.text = (
                    "Digite a data."
                )
                return

            texto_observacao = (
                observacao.text.strip()
            )

            if not texto_observacao:
                aviso.text = (
                    "Digite a observação."
                )
                return

            observacoes = (
                self.carregar_observacoes()
            )

            observacoes.append({
                "id":
                    self.proximo_id_observacao(),

                "discipulo_id":
                    discipulo.get("id"),

                "discipulo_nome":
                    discipulo.get(
                        "nome",
                        ""
                    ),

                "data":
                    data_texto,

                "texto":
                    texto_observacao,
            })

            self.salvar_observacoes(
                observacoes
            )

            popup.dismiss()

            self.status.text = (
                "Observação salva."
            )

            ao_salvar()

        salvar.bind(
            on_release=confirmar
        )

        popup.open()

    def calcular_progresso_discipulo(
        self,
        discipulo
    ):
        discipulo_id = discipulo.get(
            "id"
        )

        estudos_ativos = [
            estudo
            for estudo in self.carregar_estudos()
            if estudo.get(
                "status",
                "Ativo"
            ) == "Ativo"
        ]

        ids_estudos_ativos = {
            estudo.get("id")
            for estudo in estudos_ativos
        }

        questoes_ativas = [
            questao
            for questao in self.carregar_questoes()
            if (
                questao.get(
                    "status",
                    "Ativa"
                ) == "Ativa"
                and questao.get(
                    "estudo_id"
                ) in ids_estudos_ativos
            )
        ]

        respostas = [
            resposta
            for resposta in self.carregar_respostas()
            if resposta.get(
                "discipulo_id"
            ) == discipulo_id
        ]

        questoes_respondidas_ids = {
            resposta.get(
                "questao_id"
            )
            for resposta in respostas
            if str(
                resposta.get(
                    "resposta",
                    ""
                )
            ).strip()
        }

        questoes_ativas_ids = {
            questao.get("id")
            for questao in questoes_ativas
        }

        respondidas_ativas = (
            questoes_respondidas_ids
            & questoes_ativas_ids
        )

        total_questoes = len(
            questoes_ativas
        )

        total_respondidas = len(
            respondidas_ativas
        )

        percentual_respostas = None

        if total_questoes > 0:
            percentual_respostas = round(
                total_respondidas
                * 100
                / total_questoes
            )

        estudos_avaliaveis = 0
        estudos_concluidos = 0

        for estudo in estudos_ativos:
            questoes_estudo = [
                questao
                for questao in questoes_ativas
                if questao.get(
                    "estudo_id"
                ) == estudo.get("id")
            ]

            if not questoes_estudo:
                continue

            estudos_avaliaveis += 1

            ids_questoes_estudo = {
                questao.get("id")
                for questao in questoes_estudo
            }

            if ids_questoes_estudo.issubset(
                questoes_respondidas_ids
            ):
                estudos_concluidos += 1

        percentual_estudos = None

        if estudos_avaliaveis > 0:
            percentual_estudos = round(
                estudos_concluidos
                * 100
                / estudos_avaliaveis
            )

        presencas = [
            registro
            for registro in self.carregar_presencas()
            if registro.get(
                "discipulo_id"
            ) == discipulo_id
        ]

        presentes = sum(
            1
            for registro in presencas
            if registro.get(
                "status"
            ) == "Presente"
        )

        faltas = sum(
            1
            for registro in presencas
            if registro.get(
                "status"
            ) == "Faltou"
        )

        encontros_finalizados = (
            presentes
            + faltas
        )

        percentual_frequencia = None

        if encontros_finalizados > 0:
            percentual_frequencia = round(
                presentes
                * 100
                / encontros_finalizados
            )

        metricas = [
            valor
            for valor in (
                percentual_respostas,
                percentual_estudos,
                percentual_frequencia,
            )
            if valor is not None
        ]

        progresso_geral = (
            round(
                sum(metricas)
                / len(metricas)
            )
            if metricas
            else 0
        )

        return {
            "discipulo_id":
                discipulo_id,

            "discipulo_nome":
                discipulo.get(
                    "nome",
                    ""
                ),

            "situacao":
                discipulo.get(
                    "situacao",
                    "Ativo"
                ),

            "questoes_total":
                total_questoes,

            "questoes_respondidas":
                total_respondidas,

            "percentual_respostas":
                percentual_respostas,

            "estudos_total":
                estudos_avaliaveis,

            "estudos_concluidos":
                estudos_concluidos,

            "percentual_estudos":
                percentual_estudos,

            "presentes":
                presentes,

            "faltas":
                faltas,

            "encontros_finalizados":
                encontros_finalizados,

            "percentual_frequencia":
                percentual_frequencia,

            "progresso_geral":
                progresso_geral,
        }

    def proximo_id_presenca(self):
        presencas = self.carregar_presencas()

        ids = []

        for item in presencas:
            try:
                ids.append(
                    int(
                        item.get(
                            "id",

    def preparar_data_presenca(self, data):
        discipulos = [
            item
            for item in self.carregar_discipulos()
            if item.get(
                "situacao",
                "Ativo"
            ) == "Ativo"
        ]

        presencas = self.carregar_presencas()

        alterou = False

        for discipulo in discipulos:
            existe = False

            for registro in presencas:
                if (
                    registro.get("data") == data
                    and registro.get(
                        "discipulo_id"
                    ) == discipulo.get("id")
                ):
                    existe = True
                    break

            if not existe:
                presencas.append({
                    "id":
                        self.proximo_id_presenca(),

                    "data":
                        data,

                    "discipulo_id":
                        discipulo.get("id"),

                    "discipulo_nome":
                        discipulo.get(
                            "nome",
                            ""
                        ),

                    "status":
                        "Pendente",
                })

                alterou = True

        if alterou:
            self.salvar_presencas(
                presencas
            )

    def obter_status_presenca(
        self,
        data,
        discipulo_id
    ):
        for registro in self.carregar_presencas():
            if (
                registro.get("data") == data
                and registro.get(
                    "discipulo_id"
                ) == discipulo_id
            ):
                return registro.get(
                    "status",
                    "Pendente"
                )

        return "Pendente"

    def alterar_status_presenca(
        self,
        data,
        discipulo
    ):
        presencas = self.carregar_presencas()

        atual = "Pendente"

        registro_alvo = None

        for registro in presencas:
            if (
                registro.get("data") == data
                and registro.get(
                    "discipulo_id"
                ) == discipulo.get("id")
            ):
                registro_alvo = registro

                atual = registro.get(
                    "status",
                    "Pendente"
                )

                break

        ordem = {
            "Pendente":
                "Presente",

            "Presente":
                "Faltou",

            "Faltou":
                "Pendente",
        }

        novo_status = ordem.get(
            atual,
            "Pendente"
        )

        if registro_alvo is None:
            presencas.append({
                "id":
                    self.proximo_id_presenca(),

                "data":
                    data,

                "discipulo_id":
                    discipulo.get("id"),

                "discipulo_nome":
                    discipulo.get(
                        "nome",
                        ""
                    ),

                "status":
                    novo_status,
            })

        else:
            registro_alvo[
                "discipulo_nome"
            ] = discipulo.get(
                "nome",
                ""
            )

            registro_alvo[
                "status"
            ] = novo_status

        self.salvar_presencas(
            presencas
        )

        return novo_status

    def datas_presenca(self):
        datas = []

        for registro in self.carregar_presencas():
            data = registro.get(
                "data",
                ""
            ).strip()

            if (
                data
                and data not in datas
            ):
                datas.append(data)

        return list(
            reversed(datas)
        )

        def carregar_existente():
            discipulo = selecao[
                "discipulo"
            ]

            questao = selecao[
                "questao"
            ]

            if (
                discipulo is None
                or questao is None
            ):
                return

            for item in self.carregar_respostas():
                if (
                    item.get(
                        "discipulo_id"
                    )
                    == discipulo.get("id")
                    and item.get(
                        "questao_id"
                    )
                    == questao.get("id")
                ):
                    resposta.text = item.get(
                        "resposta",

                    )

                    aviso.text = (
                        "Resposta existente carregada."
                    )
                    return

            resposta.text = ""
            aviso.text = ""

            selecao["discipulo"] = item

            discipulo_texto.text = (
                item.get(
                    "nome",
                    ""
                )
            )

            carregar_existente()

        def recebeu_questao(item):
            selecao["questao"] = item

            pergunta = item.get(
                "pergunta",
                ""
            )

            if len(pergunta) > 70:
                pergunta = (
                    pergunta[:67]
                    + "..."
                )

            questao_texto.text = (

            self.escolher_questao_resposta(

        def confirmar(*_):
            discipulo = selecao[
                "discipulo"
            ]

            questao = selecao[
                "questao"
            ]

            if discipulo is None:
                aviso.text = (
                    "Escolha o discípulo."

            if questao is None:
                aviso.text = (
                    "Escolha a questão."
                )
                return

            resposta_texto = (
                resposta.text.strip()
            )

            if not resposta_texto:
                aviso.text = (
                    "Digite a resposta."
                )
                return

            respostas = (
                self.carregar_respostas()
            )

            existente = None

            for item in respostas:
                if (
                    item.get(
                        "discipulo_id"
                    )
                    == discipulo.get("id")
                    and item.get(
                        "questao_id"
                    )
                    == questao.get("id")
                ):
                    existente = item
                    break

            if existente is None:
                respostas.append({
                    "id":
                        self.proximo_id_resposta(),

                    "discipulo_id":
                        discipulo.get("id"),

                    "discipulo_nome":
                        discipulo.get(
                            "nome",
                            ""
                        ),

                    "questao_id":
                        questao.get("id"),

                    "estudo_id":
                        questao.get(
                            "estudo_id"
                        ),

                    "estudo_titulo":
                        questao.get(
                            "estudo_titulo",
                            ""
                        ),

                    "pergunta":
                        questao.get(
                            "pergunta",
                            ""
                        ),

                    "resposta":
                        resposta_texto,
                })

            else:
                existente[
                    "discipulo_nome"
                ] = discipulo.get(
                    "nome",
                    ""
                )

                existente[
                    "estudo_id"
                ] = questao.get(
                    "estudo_id"
                )

                existente[
                    "estudo_titulo"
                ] = questao.get(
                    "estudo_titulo",
                    ""
                )

                existente[
                    "pergunta"
                ] = questao.get(
                    "pergunta",
                    ""
                )

                existente[
                    "resposta"
                ] = resposta_texto

            self.salvar_respostas(
                respostas
            )

            popup.dismiss()

            self.status.text = (
                "Resposta salva."
            )

            ao_salvar()

        salvar.bind(
            on_release=confirmar
        )

        popup.open()

    def carregar_questoes(self):
        if not self.arquivo_questoes.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_questoes.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_questoes(self, questoes):
        self.arquivo_questoes.write_text(
            json.dumps(
                questoes,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def proximo_id_questao(self):
        questoes = self.carregar_questoes()

        ids = []

        for item in questoes:
            try:
                ids.append(
                    int(
                        item.get(
                            "id",
                            0
                        )
                    )
                )
            except Exception:
                pass

        return max(
            ids,
            default=0
        ) + 1

    def proximo_id_estudo(self):
        estudos = self.carregar_estudos()

        ids = []

        for item in estudos:
            try:
                ids.append(
                    int(

                            "id",
                            0
                        )
                    )
                )
            except Exception:

        return max(

            default=0
        ) + 1

    def paginar_texto_estudo(
        self,
        texto,
        limite=650
    ):
        texto = (
            texto
            or ""
        ).strip()

        if not texto:
            return [
                "Nenhum conteúdo cadastrado."
            ]

        palavras = texto.split()

        paginas = []
        atual = []

        tamanho = 0

        for palavra in palavras:
            adicional = (
                len(palavra)
                + (
                    1
                    if atual
                    else 0
                )
            )

            if (
                atual
                and tamanho + adicional
                > limite
            ):
                paginas.append(
                    " ".join(atual)
                )

                atual = [
                    palavra
                ]

                tamanho = len(
                    palavra
                )

            else:
                atual.append(
                    palavra
                )

                tamanho += adicional

        if atual:
            paginas.append(
                " ".join(atual)
            )

        return (
            paginas
            or [
                "Nenhum conteúdo cadastrado."
            ]
        )

        atualizar()
        popup.open()

    def novo_discipulo(self, ao_salvar):
        caixa = BoxLayout(
            orientation="vertical",
            spacing=dp(6),
            padding=dp(10)
        )

        nome = TextInput(
            hint_text="Nome",
            multiline=False,
            size_hint_y=None,
            height=dp(48)
        )

        contato = TextInput(
            hint_text="Contato",
            multiline=False,
            size_hint_y=None,
            height=dp(48)
        )

        entrada = TextInput(
            hint_text="Data de entrada",
            multiline=False,
            size_hint_y=None,
            height=dp(48)
        )

        observacoes = TextInput(
            hint_text="Observações",
            multiline=True,
            size_hint_y=None,
            height=dp(90)
        )

        caixa.add_widget(nome)
        caixa.add_widget(contato)
        caixa.add_widget(entrada)
        caixa.add_widget(observacoes)

        aviso = Label(
            text="",
            size_hint_y=None,
            height=dp(30)
        )

        caixa.add_widget(
            aviso
        )

        botoes = BoxLayout(
            spacing=dp(8),
            size_hint_y=None,
            height=dp(50)
        )

        cancelar = Button(
            text="CANCELAR"
        )

        salvar = Button(
            text="SALVAR"
        )

        botoes.add_widget(
            cancelar
        )

        botoes.add_widget(
            salvar
        )

        caixa.add_widget(
            botoes
        )

        popup = Popup(
            title="Novo Discípulo",
            content=caixa,
            size_hint=(0.94, 0.88),
            auto_dismiss=False
        )

        cancelar.bind(
            on_release=popup.dismiss
        )

        def confirmar(*_):
            nome_texto = (
                nome.text.strip()
            )

            if not nome_texto:
                aviso.text = (
                    "Digite o nome."
                )
                return

            discipulos = (
                self.carregar_discipulos()
            )

            discipulos.append({
                "id":
                    self.proximo_id_discipulo(),

                "nome":
                    nome_texto,

                "contato":
                    contato.text.strip(),

                "data_entrada":
                    entrada.text.strip(),

                "situacao":
                    "Ativo",

                "observacoes":
                    observacoes.text.strip(),
            })

            self.salvar_discipulos(
                discipulos
            )

            popup.dismiss()

            self.status.text = (
                "Discípulo salvo: "
                + nome_texto
            )

            ao_salvar()

        salvar.bind(
            on_release=confirmar
        )

        popup.open()

    def visualizar_discipulo(
        self,
        discipulo,
        ao_atualizar
    ):
        caixa = BoxLayout(
            orientation="vertical",
            spacing=dp(8),
            padding=dp(12)
        )

        caixa.add_widget(
            Label(
                text=discipulo.get(
                    "nome",
                    ""
                ),
                size_hint_y=None,
                height=dp(45),
                font_size="19sp"
            )
        )

        caixa.add_widget(
            Label(
                text=(
                    "Contato: "
                    + discipulo.get(
                        "contato",
                        ""
                    )
                ),
                size_hint_y=None,
                height=dp(38)
            )
        )

        caixa.add_widget(
            Label(
                text=(
                    "Entrada: "
                    + discipulo.get(
                        "data_entrada",
                        ""
                    )
                ),
                size_hint_y=None,
                height=dp(38)
            )
        )

        caixa.add_widget(
            Label(
                text=(
                    "Situação: "
                    + discipulo.get(
                        "situacao",
                        "Ativo"
                    )
                ),
                size_hint_y=None,
                height=dp(38)
            )
        )

        observacoes = discipulo.get(
            "observacoes",
            ""
        )

        if len(observacoes) > 180:
            observacoes = (
                observacoes[:177]
                + "..."
            )

        caixa.add_widget(
            Label(
                text=(
                    "Observações:\n"
                    + observacoes
                )
            )
        )

        botoes = BoxLayout(
            spacing=dp(8),
            size_hint_y=None,
            height=dp(50)
        )

        situacao = Button(
            text=(
                "DESATIVAR"
                if discipulo.get(
                    "situacao",
                    "Ativo"
                ) == "Ativo"
                else "ATIVAR"
            )
        )

        fechar = Button(
            text="FECHAR"
        )

        botoes.add_widget(
            situacao
        )

        botoes.add_widget(
            fechar
        )

        caixa.add_widget(
            botoes
        )

        popup = Popup(
            title="Discípulo",
            content=caixa,
            size_hint=(0.92, 0.80),
            auto_dismiss=False
        )

        def alternar(*_):
            discipulos = (
                self.carregar_discipulos()
            )

            id_alvo = discipulo.get(
                "id"
            )

            for item in discipulos:
                if item.get("id") == id_alvo:
                    item["situacao"] = (
                        "Inativo"
                        if item.get(
                            "situacao",
                            "Ativo"
                        ) == "Ativo"
                        else "Ativo"
                    )

                    discipulo["situacao"] = (
                        item["situacao"]
                    )

                    break

            self.salvar_discipulos(
                discipulos
            )

            popup.dismiss()
            ao_atualizar()

        situacao.bind(
            on_release=alternar
        )

        fechar.bind(
            on_release=popup.dismiss
        )

        popup.open()

    def build(self):
        self.title = NOME_APP
        return DiscipuladoApp()


if __name__ == "__main__":
    AplicativoGerado().run()
