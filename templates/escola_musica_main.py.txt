def obter_pasta_dados_app():
    nome_pasta_app = "".join(
        caractere
        if caractere.isalnum()
        else "_"
        for caractere in NOME_APP
    ).strip("_").lower()

    pasta = (
        Path(
            App.get_running_app().user_data_dir
        )
        / nome_pasta_app
    )

    pasta.mkdir(
        parents=True,
        exist_ok=True
    )

    return pasta

    def selecionar_modulo(self, modulo):
        if modulo == "Alunos":
            self.abrir_alunos()
            return

        if modulo == "Presença":
            self.abrir_presenca()
            return

        if modulo == "Aulas":
            self.abrir_aulas()
            return

        if modulo == "Exercícios":
            self.abrir_exercicios()
            return

        if modulo == "Progresso":
            self.abrir_progresso()
            return

        if modulo == "Instrumentos":
            self.abrir_instrumentos()
            return

        if modulo == "Repertório":
            self.abrir_repertorio()
            return

        if modulo == "Apostilas":
            self.abrir_apostilas()
            return

        self.status.text = (
            "Módulo selecionado: "
            + modulo
        )

    def carregar_alunos(self):
        if not self.arquivo_alunos.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_alunos.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_alunos(self, alunos):
        self.arquivo_alunos.write_text(
            json.dumps(
                alunos,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def abrir_alunos(self):
        caixa = BoxLayout(
            orientation="vertical",
            spacing=dp(8),
            padding=dp(10)
        )

        titulo = Label(
            text="ALUNOS",
            size_hint_y=None,
            height=dp(35),
            font_size="18sp"
        )

        caixa.add_widget(titulo)

        lista = BoxLayout(
            orientation="vertical",
            spacing=dp(6)
        )

        caixa.add_widget(lista)

        navegacao = BoxLayout(
            spacing=dp(6),
            size_hint_y=None,
            height=dp(48)
        )

        anterior = Button(
            text="ANTERIOR"
        )

        novo = Button(
            text="NOVO ALUNO"
        )

        proximo = Button(
            text="PRÓXIMO"
        )

        navegacao.add_widget(anterior)
        navegacao.add_widget(novo)
        navegacao.add_widget(proximo)

        caixa.add_widget(navegacao)

        fechar = Button(
            text="FECHAR",
            size_hint_y=None,
            height=dp(48)
        )

        caixa.add_widget(fechar)

        popup = Popup(
            title="Alunos",
            content=caixa,
            size_hint=(0.96, 0.90),
            auto_dismiss=False
        )

        def atualizar():
            lista.clear_widgets()

            alunos = self.carregar_alunos()

            por_pagina = 4

            total_paginas = max(
                1,
                (
                    len(alunos)
                    + por_pagina - 1
                ) // por_pagina
            )

            if self.pagina_alunos >= total_paginas:
                self.pagina_alunos = (
                    total_paginas - 1
                )

            inicio = (
                self.pagina_alunos
                * por_pagina
            )

            fim = inicio + por_pagina

            pagina = alunos[inicio:fim]

            if not pagina:
                lista.add_widget(
                    Label(
                        text="Nenhum aluno cadastrado."
                    )
                )

            for aluno in pagina:
                texto_aluno = (
                    aluno.get("nome", "")
                    + "\n"
                    + aluno.get(
                        "instrumento",
                        ""
                    )
                    + " - "
                    + aluno.get(
                        "situacao",
                        "Ativo"
                    )
                )

                lista.add_widget(
                    Button(
                        text=texto_aluno,
                        size_hint_y=None,
                        height=dp(65)
                    )
                )

            titulo.text = (
                "ALUNOS  "
                + str(
                    self.pagina_alunos + 1
                )
                + "/"
                + str(total_paginas)
                + "   Total: "
                + str(len(alunos))
            )

        def pagina_anterior(*_):
            if self.pagina_alunos > 0:
                self.pagina_alunos -= 1
                atualizar()

        def pagina_proxima(*_):
            alunos = self.carregar_alunos()

            total_paginas = max(
                1,
                (
                    len(alunos) + 3
                ) // 4
            )

            if (
                self.pagina_alunos + 1
                < total_paginas
            ):
                self.pagina_alunos += 1
                atualizar()

        anterior.bind(
            on_release=pagina_anterior
        )

        proximo.bind(
            on_release=pagina_proxima
        )

        novo.bind(
            on_release=lambda *_:
            self.novo_aluno(atualizar)
        )

        fechar.bind(
            on_release=popup.dismiss
        )

        atualizar()
        popup.open()

        apostilas = self.carregar_apostilas()

            0 <= indice < len(apostilas)
        ):
            return

        atual = apostilas[indice].get(
            "status",

            "Arquivada"

    def nova_apostila(self, ao_salvar):
        caixa = BoxLayout(
            orientation="vertical",

        )

        titulo = TextInput(
            hint_text="Título da apostila",

            hint_text="Instrumento",
            readonly=True,
            multiline=False,
            size_hint_y=None,

        nivel = TextInput(
            hint_text="Ex.: Iniciante / Turma A",
            multiline=False,
            size_hint_y=None,
            height=dp(48)
        )

        caixa.add_widget(
            Label(
                text="Título",
                size_hint_y=None,
                height=dp(24)
            )
        )

        caixa.add_widget(titulo)

        caixa.add_widget(
            Label(
                text="Instrumento",
                size_hint_y=None,
                height=dp(24)
            )
        )

        linha_instrumento = BoxLayout(
            spacing=dp(6),
            size_hint_y=None,
            height=dp(48)
        )

        escolher = Button(
            text="ESCOLHER",
            size_hint_x=None,
            width=dp(125)
        )

        linha_instrumento.add_widget(
            instrumento
        )

        linha_instrumento.add_widget(
            escolher
        )

        caixa.add_widget(
            linha_instrumento
        )

        caixa.add_widget(
            Label(
                text="Nível / Turma",
                size_hint_y=None,
                height=dp(24)
            )
        )

        caixa.add_widget(nivel)

        aviso = Label(
            text="",
            size_hint_y=None,
            height=dp(30)
        )

        caixa.add_widget(aviso)

        botoes = BoxLayout(
            spacing=dp(8),
            size_hint_y=None,
            height=dp(50)
        )

        cancelar = Button(
            text="CANCELAR"
        )

        salvar = Button(
            text="SALVAR"
        )

        botoes.add_widget(cancelar)
        botoes.add_widget(salvar)

        caixa.add_widget(botoes)

        popup = Popup(
            title="Nova Apostila",
            content=caixa,
            size_hint=(0.94, 0.86),
            auto_dismiss=False
        )

        escolher.bind(
            on_release=lambda *_:
            self.escolher_instrumento(
                lambda valor:
                setattr(
                    instrumento,
                    "text",
                    valor
                )
            )
        )

        cancelar.bind(
            on_release=popup.dismiss
        )

        def confirmar(*_):
            titulo_texto = (
                titulo.text.strip()
            )

            instrumento_texto = (
                instrumento.text.strip()
            )

            nivel_texto = (
                nivel.text.strip()
            )

            if not titulo_texto:
                aviso.text = (
                    "Digite o título."
                )
                return

            if not instrumento_texto:
                aviso.text = (
                    "Escolha o instrumento."
                )
                return

            if not nivel_texto:
                aviso.text = (
                    "Digite o nível ou turma."
                )
                return

            apostilas = (
                self.carregar_apostilas()
            )

            apostilas.append({
                "titulo": titulo_texto,
                "instrumento":
                    instrumento_texto,
                "nivel": nivel_texto,
                "status": "Ativa",
            })

            self.salvar_apostilas(
                apostilas
            )

            popup.dismiss()

            self.status.text = (
                "Apostila cadastrada: "
                + titulo_texto
            )

            ao_salvar()

        salvar.bind(
            on_release=confirmar
        )

        popup.open()

            "Concluído"
            if atual != "Concluído"
            else "Pendente"

    def carregar_aulas(self):
        if not self.arquivo_aulas.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_aulas.read_text(
                    encoding="utf-8"
                )

            if isinstance(dados, list):

        except Exception:
            pass

        return []

        def atualizar():
            lista.clear_widgets()

            aulas = self.carregar_aulas()

            por_pagina = 4

            total_paginas = max(
                1,
                (
                    len(aulas)
                    + por_pagina - 1
                ) // por_pagina
            )

            if self.pagina_aulas >= total_paginas:
                self.pagina_aulas = (

            inicio = (

                * por_pagina

            fim = inicio + por_pagina

            pagina = aulas[inicio:fim]

            if not pagina:
                lista.add_widget(

                    )

            for aula in pagina:
                data = aula.get(

                    "turma",
                    ""
                )

                assunto = aula.get(
                    "assunto",

                    )

                "AULAS  "
                + str(
                    self.pagina_aulas + 1

                + "/"
                + str(total_paginas)
                + "   Total: "
                + str(len(aulas))
            )

                self.pagina_aulas -= 1
                atualizar()

        def pagina_proxima(*_):
            aulas = self.carregar_aulas()

            total_paginas = max(
                1,
                (
                    len(aulas) + 3
                ) // 4
            )

            if (
                self.pagina_aulas + 1
                < total_paginas
            ):
                self.pagina_aulas += 1
                atualizar()

            on_release=pagina_anterior

            on_release=pagina_proxima

            on_release=popup.dismiss

    def abrir_presenca(self):
        alunos = self.carregar_alunos()

        if not alunos:
            self.status.text = (
                "Cadastre alunos antes de lançar presença."
            )
            return

        caixa = BoxLayout(
            orientation="vertical",
            spacing=dp(7),
            padding=dp(10)
        )

        titulo = Label(
            text="PRESENÇA",
            size_hint_y=None,
            height=dp(35),
            font_size="18sp"
        )

        caixa.add_widget(titulo)

        campo_data = TextInput(
            text=datetime.now().strftime("%d/%m/%Y"),
            multiline=False,
            size_hint_y=None,
            height=dp(48)
        )

        caixa.add_widget(campo_data)

        lista = BoxLayout(
            orientation="vertical",
            spacing=dp(6)
        )

        caixa.add_widget(lista)

        navegacao = BoxLayout(
            spacing=dp(6),
            size_hint_y=None,
            height=dp(50)
        )

        anterior = Button(text="ANTERIOR")
        salvar = Button(text="SALVAR")
        proximo = Button(text="PRÓXIMO")

        navegacao.add_widget(anterior)
        navegacao.add_widget(salvar)
        navegacao.add_widget(proximo)

        caixa.add_widget(navegacao)

        fechar = Button(
            text="FECHAR",
            size_hint_y=None,
            height=dp(48)
        )

        caixa.add_widget(fechar)

        popup = Popup(
            title="Presença",
            content=caixa,
            size_hint=(0.96, 0.92),
            auto_dismiss=False
        )

        estado = {"marcacoes": {}}

        def carregar_data():
            dados = self.carregar_presencas()
            data = campo_data.text.strip()

            estado["marcacoes"] = dict(
                dados.get(data, {})
            )

        def alternar(botao, nome):
            atual = estado["marcacoes"].get(
                nome,
                "Pendente"
            )

            novo_status = (
                "Faltou"
                if atual == "Presente"
                else "Presente"
            )

            estado["marcacoes"][nome] = novo_status

            botao.text = (
                nome
                + "\n"
                + novo_status
            )

        def atualizar():
            lista.clear_widgets()

            por_pagina = 4

            total_paginas = max(
                1,
                (len(alunos) + por_pagina - 1)
                // por_pagina
            )

            if self.pagina_presenca >= total_paginas:
                self.pagina_presenca = total_paginas - 1

            inicio = self.pagina_presenca * por_pagina
            fim = inicio + por_pagina

            for aluno in alunos[inicio:fim]:
                nome = aluno.get("nome", "")

                status = estado["marcacoes"].get(
                    nome,
                    "Pendente"
                )

                botao = Button(
                    text=nome + "\n" + status,
                    size_hint_y=None,
                    height=dp(65)
                )

                botao.bind(
                    on_release=lambda b, n=nome:
                    alternar(b, n)
                )

                lista.add_widget(botao)

            titulo.text = (
                "PRESENÇA "
                + str(self.pagina_presenca + 1)
                + "/"
                + str(total_paginas)
            )

        def pagina_anterior(*_):
            if self.pagina_presenca > 0:
                self.pagina_presenca -= 1
                atualizar()

        def pagina_proxima(*_):
            total_paginas = max(
                1,
                (len(alunos) + 3) // 4
            )

            if (
                self.pagina_presenca + 1
                < total_paginas
            ):
                self.pagina_presenca += 1
                atualizar()

        def salvar_dados(*_):
            data = campo_data.text.strip()

            if not data:
                self.status.text = "Digite a data."
                return

            dados = self.carregar_presencas()

            dados[data] = dict(
                estado["marcacoes"]
            )

            self.salvar_presencas(dados)

            self.status.text = (
                "Presença salva: " + data
            )

            titulo.text = "PRESENÇA SALVA"

        anterior.bind(
            on_release=pagina_anterior
        )

        proximo.bind(
            on_release=pagina_proxima
        )

        salvar.bind(
            on_release=salvar_dados
        )

        fechar.bind(
            on_release=popup.dismiss
        )

        carregar_data()
        atualizar()
        popup.open()
