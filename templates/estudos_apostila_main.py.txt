def obter_pasta_dados_app():
    nome_pasta_app = "".join(
        caractere
        if caractere.isalnum()
        else "_"
        for caractere in NOME_APP
    ).strip("_").lower()

    pasta = (
        Path(
            App.get_running_app().user_data_dir
        )
        / nome_pasta_app
    )

    pasta.mkdir(
        parents=True,
        exist_ok=True
    )

    return pasta

    def selecionar_modulo(self, modulo):
        if modulo == "Apostilas":
            self.abrir_apostilas()
            return

        if modulo == "Lições":
            self.abrir_licoes()
            return

        if modulo == "Questões":
            self.abrir_questoes()
            return

        if modulo == "Respostas":
            self.abrir_respostas()
            return

        if modulo == "Conclusão de estudo":
            self.abrir_conclusao_estudo()
            return

        if modulo == "Progresso":
            self.abrir_progresso()
            return

    def calcular_progresso_apostila(
        self,
        apostila
    ):
        apostila_id = apostila.get("id")

        licoes = [
            l
            for l in self.carregar_licoes()
            if l.get("apostila_id")
            == apostila_id
        ]

        ids_licoes = {
            l.get("id")
            for l in licoes
        }

        questoes = [
            q
            for q in self.carregar_questoes()
            if q.get("licao_id")
            in ids_licoes
        ]

        ids_questoes = {
            q.get("id")
            for q in questoes
        }

        respostas = self.carregar_respostas()

        questoes_respondidas = {
            r.get("questao_id")
            for r in respostas
            if r.get("questao_id")
            in ids_questoes
        }

        total_licoes = len(licoes)

        licoes_concluidas = sum(
            1
            for licao in licoes
            if self.calcular_conclusao_licao(
                licao
            )["concluido"]
        )

        total_questoes = len(questoes)

        total_respondidas = len(
            questoes_respondidas
        )

        unidades_totais = (
            total_licoes
            + total_questoes
        )

        unidades_concluidas = (
            licoes_concluidas
            + total_respondidas
        )

        percentual = (
            round(
                unidades_concluidas
                * 100
                / unidades_totais
            )
            if unidades_totais
            else 0
        )

        return {
            "apostila_id":
                apostila_id,

            "titulo":
                apostila.get(
                    "titulo",
                    ""
                ),

            "total_licoes":
                total_licoes,

            "licoes_concluidas":
                licoes_concluidas,

            "total_questoes":
                total_questoes,

            "questoes_respondidas":
                total_respondidas,

            "percentual":
                percentual,
        }

    def calcular_conclusao_licao(
        self,
        licao
    ):
        licao_id = licao.get("id")

        questoes = [
            q
            for q in self.carregar_questoes()
            if q.get("licao_id") == licao_id
        ]

        respostas = self.carregar_respostas()

        respondidas = {
            r.get("questao_id")
            for r in respostas
            if r.get("questao_id") is not None
        }

        total_questoes = len(questoes)

        total_respondidas = sum(
            1
            for questao in questoes
            if questao.get("id")
            in respondidas
        )

        leitura_concluida = (
            licao.get(
                "status",
                "Em estudo"
            )
            == "Concluída"
        )

        if total_questoes == 0:
            concluido = leitura_concluida
        else:
            concluido = (
                leitura_concluida
                and total_respondidas
                == total_questoes
            )

        return {
            "licao_id": licao_id,
            "titulo": licao.get(
                "titulo",
                ""
            ),
            "apostila": licao.get(
                "apostila_titulo",
                ""
            ),
            "leitura_concluida":
                leitura_concluida,
            "total_questoes":
                total_questoes,
            "respondidas":
                total_respondidas,
            "concluido":
                concluido,
        }

    def carregar_respostas(self):
        if not self.arquivo_respostas.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_respostas.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_respostas(self, respostas):
        self.arquivo_respostas.write_text(
            json.dumps(
                respostas,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def proximo_id_resposta(self):
        respostas = self.carregar_respostas()

        ids = []

        for item in respostas:
            try:
                ids.append(
                    int(item.get("id", 0))
                )
            except Exception:
                pass

        return max(ids, default=0) + 1

    def carregar_questoes(self):
        if not self.arquivo_questoes.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_questoes.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_questoes(self, questoes):
        self.arquivo_questoes.write_text(
            json.dumps(
                questoes,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def proximo_id_questao(self):
        questoes = self.carregar_questoes()

        ids = []

        for item in questoes:
            try:
                ids.append(
                    int(item.get("id", 0))
                )
            except Exception:
                pass

        return max(ids, default=0) + 1

    def extrair_trechos_para_questoes(
        self,
        conteudo
    ):
        conteudo = re.sub(
            r"\s+",
            " ",
            conteudo or ""
        ).strip()

        if not conteudo:
            return []

        trechos = re.split(
            r"(?<=[.!?])\s+",
            conteudo
        )

        limpos = []

        for trecho in trechos:
            trecho = trecho.strip()

            if len(trecho) >= 35:
                limpos.append(trecho)

        if not limpos:
            palavras = conteudo.split()

            bloco = []

            for palavra in palavras:
                bloco.append(palavra)

                if len(" ".join(bloco)) >= 120:
                    limpos.append(
                        " ".join(bloco)
                    )
                    bloco = []

            if bloco:
                limpos.append(
                    " ".join(bloco)
                )

        return limpos

    def gerar_questoes_do_texto(
        self,
        licao,
        quantidade=5
    ):
        conteudo = licao.get(
            "conteudo",
            ""
        )

        trechos = (
            self.extrair_trechos_para_questoes(
                conteudo
            )
        )

        if not trechos:
            return 0

        questoes = (
            self.carregar_questoes()
        )

        licao_id = licao.get("id")

        fontes_existentes = {
            q.get("trecho_fonte")
            for q in questoes
            if (
                q.get("licao_id")
                == licao_id
                and q.get("trecho_fonte")
            )
        }

        proximo_id = (
            self.proximo_id_questao()
        )

        criadas = 0

        modelos = [
            (
                "Explique com suas palavras "
                "a ideia principal deste trecho: "
            ),
            (
                "O que o trecho a seguir "
                "ensina? "
            ),
            (
                "Qual é o ponto principal "
                "apresentado nesta afirmação? "
            ),
        ]

        for indice, trecho in enumerate(
            trechos
        ):
            if criadas >= quantidade:
                break

            if trecho in fontes_existentes:
                continue

            resumo = trecho

            if len(resumo) > 180:
                resumo = (
                    resumo[:177]
                    + "..."
                )

            inicio = modelos[
                criadas % len(modelos)
            ]

            pergunta = (
                inicio
                + '"'
                + resumo
                + '"'
            )

            questoes.append({
                "id":
                    proximo_id,

                "licao_id":
                    licao_id,

                "licao_titulo":
                    licao.get(
                        "titulo",
                        ""
                    ),

                "pergunta":
                    pergunta,

                "origem":
                    "texto_da_licao",

                "trecho_fonte":
                    trecho,
            })

            proximo_id += 1
            criadas += 1

        if criadas:
            self.salvar_questoes(
                questoes
            )

        return criadas

    def proximo_id_licao(self):
        licoes = self.carregar_licoes()

        ids = []

        for item in licoes:
            try:
                ids.append(
                    int(item.get("id", 0))
                )
            except Exception:
                pass

        return max(ids, default=0) + 1

    def paginar_texto(
        self,
        texto,
        limite=700
    ):
        palavras = texto.split()

        if not palavras:
            return ["Sem conteúdo."]

        paginas = []
        atual = ""

        for palavra in palavras:
            tentativa = (
                atual + " " + palavra
            ).strip()

            if (
                len(tentativa) > limite
                and atual
            ):
                paginas.append(atual)
                atual = palavra
            else:
                atual = tentativa

        if atual:
            paginas.append(atual)

        return paginas or ["Sem conteúdo."]

        for item in apostilas:
            try:
                ids.append(
                    int(item.get("id", 0))
                )
            except Exception:
                pass

        return max(ids, default=0) + 1

    def alternar_status_apostila(self, indice):
        apostilas = self.carregar_apostilas()

        if not (
            0 <= indice < len(apostilas)
        ):
            return

        atual = apostilas[
            indice
        ].get(
            "status",
            "Ativa"
        )

        apostilas[
            indice
        ]["status"] = (
            "Arquivada"
            if atual == "Ativa"
            else "Ativa"
        )

        self.salvar_apostilas(
            apostilas
        )

    def abrir_apostilas(self):
        caixa = BoxLayout(
            orientation="vertical",
            spacing=dp(7),
            padding=dp(10)
        )

        titulo = Label(
            text="APOSTILAS",
            size_hint_y=None,
            height=dp(35),
            font_size="18sp"
        )

        caixa.add_widget(titulo)

        lista = BoxLayout(
            orientation="vertical",
            spacing=dp(6)
        )

        caixa.add_widget(lista)

        navegacao = BoxLayout(
            spacing=dp(6),
            size_hint_y=None,
            height=dp(50)
        )

        anterior = Button(
            text="ANTERIOR"
        )

        nova = Button(
            text="NOVA"
        )

        proximo = Button(
            text="PRÓXIMO"
        )

        navegacao.add_widget(anterior)
        navegacao.add_widget(nova)
        navegacao.add_widget(proximo)

        caixa.add_widget(navegacao)

        fechar = Button(
            text="FECHAR",
            size_hint_y=None,
            height=dp(48)
        )

        caixa.add_widget(fechar)

        popup = Popup(
            title="Apostilas",
            content=caixa,
            size_hint=(0.96, 0.92),
            auto_dismiss=False
        )

        def atualizar():
            lista.clear_widgets()

            apostilas = (
                self.carregar_apostilas()
            )

            por_pagina = 4

            total_paginas = max(
                1,
                (
                    len(apostilas)
                    + por_pagina - 1
                ) // por_pagina
            )

            if (
                self.pagina_apostilas
                >= total_paginas
            ):
                self.pagina_apostilas = (
                    total_paginas - 1
                )

            inicio = (
                self.pagina_apostilas
                * por_pagina
            )

            fim = inicio + por_pagina

            pagina = apostilas[
                inicio:fim
            ]

            if not pagina:
                lista.add_widget(
                    Label(
                        text=(
                            "Nenhuma apostila cadastrada."
                        )
                    )
                )

            for posicao, item in enumerate(
                pagina
            ):
                indice_real = (
                    inicio + posicao
                )

                nome = item.get(
                    "titulo",
                    ""
                )

                status = item.get(
                    "status",
                    "Ativa"
                )

                identificador = item.get(
                    "id",
                    ""
                )

                botao = Button(
                    text=(
                        nome
                        + " - "
                        + status
                        + "\n"
                        + "ID: "
                        + str(identificador)
                    ),
                    size_hint_y=None,
                    height=dp(65)
                )

                def mudar(
                    _botao,
                    indice=indice_real
                ):
                    self.alternar_status_apostila(
                        indice
                    )
                    atualizar()

                botao.bind(
                    on_release=mudar
                )

                lista.add_widget(
                    botao
                )

            titulo.text = (
                "APOSTILAS  "
                + str(
                    self.pagina_apostilas + 1
                )
                + "/"
                + str(total_paginas)
                + "   Total: "
                + str(len(apostilas))
            )

        def pagina_anterior(*_):
            if self.pagina_apostilas > 0:
                self.pagina_apostilas -= 1
                atualizar()

        def pagina_proxima(*_):
            apostilas = (
                self.carregar_apostilas()
            )

            total_paginas = max(
                1,
                (
                    len(apostilas) + 3
                ) // 4
            )

            if (
                self.pagina_apostilas + 1
                < total_paginas
            ):
                self.pagina_apostilas += 1
                atualizar()

        anterior.bind(
            on_release=pagina_anterior
        )

        proximo.bind(
            on_release=pagina_proxima
        )

        nova.bind(
            on_release=lambda *_:
            self.nova_apostila(
                atualizar
            )
        )

        fechar.bind(
            on_release=popup.dismiss
        )

        atualizar()
        popup.open()

    def nova_apostila(self, ao_salvar):
        caixa = BoxLayout(
            orientation="vertical",
            spacing=dp(8),
            padding=dp(10)
        )

        titulo = TextInput(
            hint_text="Título da apostila",
            multiline=False,
            size_hint_y=None,
            height=dp(50)
        )

        descricao = TextInput(
            hint_text="Descrição da apostila",
            multiline=True,
            size_hint_y=None,
            height=dp(100)
        )

        caixa.add_widget(
            Label(
                text="Título",
                size_hint_y=None,
                height=dp(28)
            )
        )

        caixa.add_widget(titulo)

        caixa.add_widget(
            Label(
                text="Descrição",
                size_hint_y=None,
                height=dp(28)
            )
        )

        caixa.add_widget(descricao)

        aviso = Label(
            text="",
            size_hint_y=None,
            height=dp(32)
        )

        caixa.add_widget(aviso)

        botoes = BoxLayout(
            spacing=dp(8),
            size_hint_y=None,
            height=dp(52)
        )

        cancelar = Button(
            text="CANCELAR"
        )

        salvar = Button(
            text="SALVAR"
        )

        botoes.add_widget(cancelar)
        botoes.add_widget(salvar)

        caixa.add_widget(botoes)

        popup = Popup(
            title="Nova Apostila",
            content=caixa,
            size_hint=(0.92, 0.80),
            auto_dismiss=False
        )

        cancelar.bind(
            on_release=popup.dismiss
        )

        def confirmar(*_):
            titulo_texto = (
                titulo.text.strip()
            )

            if not titulo_texto:
                aviso.text = (
                    "Digite o título da apostila."
                )
                return

            apostilas = (
                self.carregar_apostilas()
            )

            apostilas.append({
                "id":
                    self.proximo_id_apostila(),

                "titulo":
                    titulo_texto,

                "descricao":
                    descricao.text.strip(),

                "status":
                    "Ativa",
            })

            self.salvar_apostilas(
                apostilas
            )

            popup.dismiss()

            self.status.text = (
                "Apostila cadastrada: "
                + titulo_texto
            )

            ao_salvar()

        salvar.bind(
            on_release=confirmar
        )

        popup.open()


class AplicativoGerado(App):

    def build(self):
        self.title = NOME_APP
        return EstudosApostila()


if __name__ == "__main__":
    AplicativoGerado().run()
