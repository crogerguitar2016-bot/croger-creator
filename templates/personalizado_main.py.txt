    def selecionar_modulo(
        self,
        modulo
    ):
        if modulo == "Biblioteca":
            self.abrir_biblioteca()
            return

        if modulo == "Apostilas":
            self.abrir_apostilas()
            return

        if modulo == "Questões":
            self.abrir_questoes()
            return

        if modulo == "Cadastros":
            self.abrir_cadastros()
            return

        if modulo == "Planilhas":
            self.abrir_planilhas()
            return

        if modulo == "Presença":
            self.abrir_presencas()
            return

        if modulo == "Exercícios":
            self.abrir_exercicios()
            return

        if modulo == "Progresso":
            self.abrir_progresso()
            return

        if modulo == "Relatórios":
            self.abrir_relatorios()
            return

        self.status.text = (
            "Módulo selecionado: "
            + modulo
        )


    def calcular_relatorio_geral(self):
        biblioteca = (
            self.carregar_biblioteca()
        )

        apostilas = (
            self.carregar_apostilas()
        )

        questoes = (
            self.carregar_questoes()
        )

        cadastros = (
            self.carregar_cadastros()
        )

        exercicios = (
            self.carregar_exercicios()
        )

        presencas = (
            self.carregar_presencas()
        )

        # ------------------------------------------
        # Biblioteca
        # ------------------------------------------

        biblioteca_ativos = sum(
            1
            for item in biblioteca
            if item.get(
                "status",
                "Ativo"
            ) == "Ativo"
        )

        biblioteca_arquivados = sum(
            1
            for item in biblioteca
            if item.get(
                "status"
            ) == "Arquivado"
        )

        # ------------------------------------------
        # Apostilas
        # ------------------------------------------

        apostilas_ativas = sum(
            1
            for item in apostilas
            if item.get(
                "status",
                "Ativa"
            ) == "Ativa"
        )

        apostilas_arquivadas = sum(
            1
            for item in apostilas
            if item.get(
                "status"
            ) == "Arquivada"
        )

        # ------------------------------------------
        # Questões
        # ------------------------------------------

        questoes_ativas = sum(
            1
            for item in questoes
            if item.get(
                "status",
                "Ativa"
            ) == "Ativa"
        )

        questoes_arquivadas = sum(
            1
            for item in questoes
            if item.get(
                "status"
            ) == "Arquivada"
        )

        # ------------------------------------------
        # Cadastros
        # ------------------------------------------

        cadastros_ativos = sum(
            1
            for item in cadastros
            if item.get(
                "status",
                "Ativo"
            ) == "Ativo"
        )

        cadastros_inativos = sum(
            1
            for item in cadastros
            if item.get(
                "status"
            ) == "Inativo"
        )

        # ------------------------------------------
        # Categorias
        # ------------------------------------------

        categorias = {}

        for cadastro in cadastros:
            categoria = str(
                cadastro.get(
                    "categoria",
                    ""
                )
            ).strip()

            if not categoria:
                categoria = (
                    "Sem categoria"
                )

            categorias[
                categoria
            ] = (
                categorias.get(
                    categoria,
                    0
                )
                + 1
            )

        # ------------------------------------------
        # Exercícios
        # ------------------------------------------

        exercicios_concluidos = sum(
            1
            for item in exercicios
            if item.get(
                "status"
            ) == "Concluído"
        )

        exercicios_pendentes = sum(
            1
            for item in exercicios
            if item.get(
                "status",
                "Pendente"
            ) == "Pendente"
        )

        if exercicios:
            percentual_exercicios = (
                exercicios_concluidos
                / len(exercicios)
                * 100
            )
        else:
            percentual_exercicios = None

        # ------------------------------------------
        # Presença
        # ------------------------------------------

        presentes = sum(
            1
            for item in presencas
            if item.get(
                "status"
            ) == "Presente"
        )

        faltas = sum(
            1
            for item in presencas
            if item.get(
                "status"
            ) == "Faltou"
        )

        pendentes = sum(
            1
            for item in presencas
            if item.get(
                "status",
                "Pendente"
            ) == "Pendente"
        )

        total_frequencia = (
            presentes
            + faltas
        )

        if total_frequencia:
            frequencia_geral = (
                presentes
                / total_frequencia
                * 100
            )
        else:
            frequencia_geral = None

        # ------------------------------------------
        # Progresso médio
        # ------------------------------------------

        progressos_validos = []

        for cadastro in cadastros:
            dados = (
                self.calcular_progresso_cadastro(
                    cadastro
                )
            )

            valor = dados.get(
                "progresso_geral"
            )

            if valor is not None:
                progressos_validos.append(
                    valor
                )

        if progressos_validos:
            progresso_medio = (
                sum(
                    progressos_validos
                )
                / len(
                    progressos_validos
                )
            )
        else:
            progresso_medio = None

        return {
            "biblioteca_total":
                len(biblioteca),

            "biblioteca_ativos":
                biblioteca_ativos,

            "biblioteca_arquivados":
                biblioteca_arquivados,

            "apostilas_total":
                len(apostilas),

            "apostilas_ativas":
                apostilas_ativas,

            "apostilas_arquivadas":
                apostilas_arquivadas,

            "questoes_total":
                len(questoes),

            "questoes_ativas":
                questoes_ativas,

            "questoes_arquivadas":
                questoes_arquivadas,

            "cadastros_total":
                len(cadastros),

            "cadastros_ativos":
                cadastros_ativos,

            "cadastros_inativos":
                cadastros_inativos,

            "categorias":
                categorias,

            "exercicios_total":
                len(exercicios),

            "exercicios_concluidos":
                exercicios_concluidos,

            "exercicios_pendentes":
                exercicios_pendentes,

            "percentual_exercicios":
                percentual_exercicios,

            "presencas_total":
                len(presencas),

            "presentes":
                presentes,

            "faltas":
                faltas,

            "pendentes":
                pendentes,

            "frequencia_geral":
                frequencia_geral,

            "progresso_medio":
                progresso_medio,
        }

        def obter_paginas():
            linhas = (
                self.montar_linhas_relatorio()
            )

            por_pagina = 8

            paginas = []

            for inicio in range(
                0,
                len(linhas),
                por_pagina
            ):
                paginas.append(
                    linhas[
                        inicio:
                        inicio + por_pagina
                    ]
                )

            if not paginas:
                paginas = [[
                    "Nenhum dado disponível."
                ]]

            return paginas

        def atualizar():
            paginas = (
                obter_paginas()
            )

            if (
                self.pagina_relatorios
                >= len(paginas)
            ):
                self.pagina_relatorios = (
                    len(paginas) - 1
                )

            if self.pagina_relatorios < 0:
                self.pagina_relatorios = 0

                "\n\n".join(

            if self.pagina_relatorios > 0:
                self.pagina_relatorios -= 1

        def avancar(*_):
            paginas = (
                obter_paginas()

            if (
                self.pagina_relatorios + 1
                < len(paginas)
            ):

                atualizar()

            on_release=voltar
        )

        proximo.bind(
            on_release=avancar
        )

        fechar.bind(
            on_release=popup.dismiss
        )

        popup.open()

    def calcular_progresso_cadastro(
        self,
        cadastro
    ):
        cadastro_id = cadastro.get(
            "id"
        )

        exercicios = [
            item
            for item in self.carregar_exercicios()
            if item.get(
                "cadastro_id"
            ) == cadastro_id
        ]

        total_exercicios = len(
            exercicios
        )

        concluidos = sum(
            1
            for item in exercicios
            if item.get(
                "status"
            ) == "Concluído"
        )

        if total_exercicios:
            percentual_exercicios = (
                concluidos
                / total_exercicios
                * 100
            )
        else:
            percentual_exercicios = None

        presencas_validas = [
            item
            for item in self.carregar_presencas()
            if (
                item.get(
                    "cadastro_id"
                ) == cadastro_id
                and item.get(
                    "status"
                ) in (
                    "Presente",
                    "Faltou",
                )
            )
        ]

        presentes = sum(
            1
            for item in presencas_validas
            if item.get(
                "status"
            ) == "Presente"
        )

        faltas = sum(
            1
            for item in presencas_validas
            if item.get(
                "status"
            ) == "Faltou"
        )

        total_frequencia = (
            presentes
            + faltas
        )

        if total_frequencia:
            percentual_frequencia = (
                presentes
                / total_frequencia
                * 100
            )
        else:
            percentual_frequencia = None

        indicadores = []

        if percentual_exercicios is not None:
            indicadores.append(
                percentual_exercicios
            )

        if percentual_frequencia is not None:
            indicadores.append(
                percentual_frequencia
            )

        if indicadores:
            progresso_geral = (
                sum(indicadores)
                / len(indicadores)
            )
        else:
            progresso_geral = None

        return {
            "total_exercicios":
                total_exercicios,

            "concluidos":
                concluidos,

            "percentual_exercicios":
                percentual_exercicios,

            "presentes":
                presentes,

            "faltas":
                faltas,

            "percentual_frequencia":
                percentual_frequencia,

            "progresso_geral":
                progresso_geral,
        }

    def texto_percentual(
        self,
        valor
    ):
        if valor is None:
            return "Sem dados"

        return (
            str(
                round(
                    valor,
                    1
                )
            ).replace(
                ".0",
                ""
            )
            + "%"
        )

    def abrir_progresso(self):
        caixa = BoxLayout(
            orientation="vertical",
            spacing=dp(7),
            padding=dp(10)
        )

        titulo = Label(
            text="PROGRESSO",
            size_hint_y=None,
            height=dp(38),
            font_size="18sp"
        )

        caixa.add_widget(
            titulo
        )

        lista = BoxLayout(
            orientation="vertical",
            spacing=dp(6)
        )

        caixa.add_widget(
            lista
        )

        navegacao = BoxLayout(
            spacing=dp(7),
            size_hint_y=None,
            height=dp(50)
        )

        anterior = Button(
            text="ANTERIOR"
        )

        proximo = Button(
            text="PRÓXIMO"
        )

        navegacao.add_widget(
            anterior
        )

        navegacao.add_widget(
            proximo
        )

        caixa.add_widget(
            navegacao
        )

        fechar = Button(
            text="FECHAR",
            size_hint_y=None,
            height=dp(48)
        )

        caixa.add_widget(
            fechar
        )

        popup = Popup(
            title="Progresso",
            content=caixa,
            size_hint=(0.96, 0.92),
            auto_dismiss=False
        )

        def atualizar():
            lista.clear_widgets()

            cadastros = (
                self.carregar_cadastros()
            )

            por_pagina = 4

            total_paginas = max(
                1,
                (
                    len(cadastros)
                    + por_pagina - 1
                ) // por_pagina
            )

            if (
                self.pagina_progresso
                >= total_paginas
            ):
                self.pagina_progresso = (
                    total_paginas - 1
                )

            inicio = (
                self.pagina_progresso
                * por_pagina
            )

            fim = inicio + por_pagina

            pagina = cadastros[
                inicio:fim
            ]

            if not pagina:
                lista.add_widget(
                    Label(
                        text=(
                            "Nenhum cadastro disponível."
                        )
                    )
                )

            for cadastro in pagina:
                dados = (
                    self.calcular_progresso_cadastro(
                        cadastro
                    )
                )

                botao = Button(
                    text=(
                        cadastro.get(
                            "nome",
                            ""
                        )
                        + " - "
                        + cadastro.get(
                            "status",
                            "Ativo"
                        )
                        + "\n"
                        + "Progresso geral: "
                        + self.texto_percentual(
                            dados[
                                "progresso_geral"
                            ]
                        )
                    ),
                    size_hint_y=None,
                    height=dp(70)
                )

                botao.bind(
                    on_release=lambda _b, c=cadastro:
                    self.visualizar_progresso(
                        c
                    )
                )

                lista.add_widget(
                    botao
                )

            titulo.text = (
                "PROGRESSO  "
                + str(
                    self.pagina_progresso + 1
                )
                + "/"
                + str(
                    total_paginas
                )
                + "   Total: "
                + str(
                    len(cadastros)
                )
            )

        def voltar(*_):
            if self.pagina_progresso > 0:
                self.pagina_progresso -= 1
                atualizar()

        def avancar(*_):
            cadastros = (
                self.carregar_cadastros()
            )

            total_paginas = max(
                1,
                (
                    len(cadastros) + 3
                ) // 4
            )

            if (
                self.pagina_progresso + 1
                < total_paginas
            ):
                self.pagina_progresso += 1
                atualizar()

        anterior.bind(
            on_release=voltar
        )

        proximo.bind(

    def preparar_data_presenca(
        self,
        data
    ):
        cadastros = [
            item
            for item in self.carregar_cadastros()
            if item.get(
                "status",
                "Ativo"
            ) == "Ativo"
        ]

        presencas = (
            self.carregar_presencas()
        )

        pares_existentes = {
            (
                registro.get(
                    "data"
                ),
                registro.get(
                    "cadastro_id"
                ),
            )
            for registro in presencas
        }

        proximo_id = (
            self.proximo_id_presenca(
                presencas
            )
        )

        alterou = False

        for cadastro in cadastros:
            chave = (
                data,
                cadastro.get(
                    "id"
                ),
            )

            if chave in pares_existentes:
                continue

            presencas.append({
                "id":
                    proximo_id,

                "data":
                    data,

                "cadastro_id":
                    cadastro.get(
                        "id"
                    ),

                "cadastro_nome":
                    cadastro.get(
                        "nome",
                        ""
                    ),

                "categoria":
                    cadastro.get(
                        "categoria",
                        ""
                    ),

                "status":
                    "Pendente",
            })

            proximo_id += 1
            pares_existentes.add(
                chave
            )
            alterou = True

        if alterou:
            self.salvar_presencas(
                presencas
            )

    def alterar_status_presenca(
        self,
        data,
        cadastro_id
    ):
        presencas = (
            self.carregar_presencas()
        )

        ordem = {
            "Pendente":
                "Presente",

            "Presente":
                "Faltou",

            "Faltou":
                "Pendente",
        }

        novo_status = "Pendente"

        for registro in presencas:
            if (
                registro.get(
                    "data"
                ) == data
                and registro.get(
                    "cadastro_id"
                ) == cadastro_id
            ):
                atual = registro.get(
                    "status",
                    "Pendente"
                )

                novo_status = ordem.get(
                    atual,
                    "Pendente"
                )

                registro[
                    "status"
                ] = novo_status

                break

        self.salvar_presencas(
            presencas
        )

        return novo_status

    def datas_presenca(self):
        datas = []

        for registro in self.carregar_presencas():
            data = str(
                registro.get(
                    "data",
                    ""
                )
            ).strip()

            if (
                data
                and data not in datas
            ):
                datas.append(
                    data
                )

        return list(
            reversed(
                datas
            )
        )

    def carregar_cadastros(self):
        if not self.arquivo_cadastros.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_cadastros.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_cadastros(self, cadastros):
        self.arquivo_cadastros.write_text(
            json.dumps(
                cadastros,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def proximo_id_cadastro(self):
        cadastros = self.carregar_cadastros()

        ids = []

        for item in cadastros:
            try:
                ids.append(
                    int(
                        item.get(
                            "id",
                            0
                        )
                    )
                )
            except Exception:
                pass

        return max(
            ids,
            default=0
        ) + 1

    def paginar_texto_apostila(
        self,
        texto,
        limite=650
    ):
        texto = (
            texto
            or ""
        ).strip()

        if not texto:
            return [
                "Nenhum conteúdo cadastrado."
            ]

        palavras = texto.split()

        paginas = []
        pagina_atual = []
        tamanho = 0

        for palavra in palavras:
            adicional = (
                len(palavra)
                + (
                    1
                    if pagina_atual
                    else 0
                )
            )

            if (
                pagina_atual
                and tamanho + adicional
                > limite
            ):
                paginas.append(
                    " ".join(
                        pagina_atual
                    )
                )

                pagina_atual = [
                    palavra
                ]

                tamanho = len(
                    palavra
                )

            else:
                pagina_atual.append(
                    palavra
                )

                tamanho += adicional

        if pagina_atual:
            paginas.append(
                " ".join(
                    pagina_atual
                )
            )

        return (
            paginas
            or [
                "Nenhum conteúdo cadastrado."
            ]
        )

    def carregar_biblioteca(self):
        if not self.arquivo_biblioteca.exists():
            return []

        try:
            dados = json.loads(
                self.arquivo_biblioteca.read_text(
                    encoding="utf-8"
                )
            )

            if isinstance(dados, list):
                return dados

        except Exception:
            pass

        return []

    def salvar_biblioteca(self, itens):
        self.arquivo_biblioteca.write_text(
            json.dumps(
                itens,
                ensure_ascii=False,
                indent=2
            ),
            encoding="utf-8"
        )

    def proximo_id_biblioteca(self):
        itens = self.carregar_biblioteca()

        ids = []

        for item in itens:
            try:
                ids.append(
                    int(
                        item.get(
                            "id",
                            0
                        )
                    )
                )
            except Exception:
                pass

        return max(
            ids,
            default=0
        ) + 1
